# To check files that are created or changd in the local repo [Status: Untracked or Not Staged]

git status


# To STAGE changd files or New Files [Status: Staged]
git add <filename>

or

git add . -- This will add all files in the current directory and sub-directories (. = current directory)

# To move all staged files into LOCAL repository [Status: commit]
git commit -m "One line comment here"


# To push files from LOCAL repository to REMOTE repository [Status: PUSH]
git push
